## Requirements

### 1. Cache Available Plans

```typescript
// Plans don't change often - cache for 1 hour
const PLANS_CACHE_KEY = 'plans:all';
const PLANS_CACHE_TTL = 3600;

export const getAvailablePlans = async () => {
  const cached = await redis.get(PLANS_CACHE_KEY);
  if (cached) return JSON.parse(cached);

  const plans = PLAN_CONFIG;
  await redis.setex(PLANS_CACHE_KEY, PLANS_CACHE_TTL, JSON.stringify(plans));
  return plans;
};
```

### 2. Cache User Permissions

```typescript
// Cache user plan limits
const getUserLimits = async (userId: string) => {
  const cacheKey = `user:limits:${userId}`;
  const cached = await redis.get(cacheKey);
  if (cached) return JSON.parse(cached);

  const user = await User.findById(userId);
  const limits = getPlanLimits(user.subscription.plan);

  await redis.setex(cacheKey, 300, JSON.stringify(limits)); // 5 min
  return limits;
};
```

### 3. Response Caching for Analytics

```typescript
// Cache analytics responses
const analyticsMiddleware = async (req, res, next) => {
  const cacheKey = `analytics:${req.user._id}:${req.path}:${JSON.stringify(req.query)}`;
  const cached = await redis.get(cacheKey);

  if (cached) {
    return res.json(JSON.parse(cached));
  }

  // Store original json method
  const originalJson = res.json.bind(res);
  res.json = (data) => {
    redis.setex(cacheKey, 60, JSON.stringify(data)); // 1 min cache
    return originalJson(data);
  };

  next();
};
```

### 4. Cache Warming

```typescript
// Warm caches on startup
export const warmCaches = async () => {
  // Warm plans cache
  await getAvailablePlans();

  // Warm active users' limits
  const activeUsers = await User.find({ 'subscription.status': 'active' })
    .select('_id subscription.plan')
    .limit(1000);

  for (const user of activeUsers) {
    await getUserLimits(user._id);
  }
};
```

### 5. Cache Metrics

```typescript
// Track cache hits/misses
const cacheStats = {
  hits: 0,
  misses: 0,
  ratio: () => cacheStats.hits / (cacheStats.hits + cacheStats.misses)
};

export const getCacheStats = () => ({
  ...cacheStats,
  hitRatio: (cacheStats.ratio() * 100).toFixed(2) + '%'
});
```
