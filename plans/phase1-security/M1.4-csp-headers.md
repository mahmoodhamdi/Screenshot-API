# Milestone 1.4: Fix CSP & Security Headers

## Status: NOT_STARTED
## Priority: HIGH
## Estimated Effort: Medium

---

## Context

The current Content Security Policy (CSP) in `src/app.ts` uses `'unsafe-inline'` and `'unsafe-eval'` directives, which significantly weaken XSS protection. While these are needed for Swagger UI and ReDoc, they should not apply to the main application routes.

### Risk Level: MEDIUM-HIGH
- XSS attacks possible on application pages
- Inline script injection risk
- eval-based attacks possible

---

## Current State

### Current CSP Configuration (app.ts lines 47-72)

```typescript
app.use(
  helmet({
    contentSecurityPolicy: {
      directives: {
        defaultSrc: ["'self'"],
        styleSrc: ["'self'", "'unsafe-inline'", 'https://fonts.googleapis.com'],
        scriptSrc: ["'self'", "'unsafe-inline'", "'unsafe-eval'"],
        fontSrc: ["'self'", 'https://fonts.gstatic.com'],
        imgSrc: ["'self'", 'data:', 'https:'],
        connectSrc: ["'self'"],
      },
    },
  })
);
```

### Problems
1. `'unsafe-inline'` allows any inline scripts
2. `'unsafe-eval'` allows eval() calls
3. Same CSP for all routes (docs and app)
4. No nonce-based script loading
5. Missing some security headers

## Requirements

### 1. Create Separate CSP Configurations

```typescript
// src/config/security.ts

import crypto from 'crypto';
import { Request, Response, NextFunction } from 'express';

// Generate nonce for each request
export const generateNonce = (): string => {
  return crypto.randomBytes(16).toString('base64');
};

// Strict CSP for application routes
export const strictCSP = {
  directives: {
    defaultSrc: ["'self'"],
    styleSrc: ["'self'", 'https://fonts.googleapis.com'],
    scriptSrc: ["'self'"], // Will add nonce dynamically
    fontSrc: ["'self'", 'https://fonts.gstatic.com'],
    imgSrc: ["'self'", 'data:', 'https:'],
    connectSrc: ["'self'"],
    frameSrc: ["'none'"],
    objectSrc: ["'none'"],
    baseUri: ["'self'"],
    formAction: ["'self'"],
    upgradeInsecureRequests: [],
  },
};

// Relaxed CSP for documentation routes (Swagger, ReDoc)
export const docsCSP = {
  directives: {
    defaultSrc: ["'self'"],
    styleSrc: ["'self'", "'unsafe-inline'", 'https://fonts.googleapis.com', 'https://cdn.redoc.ly'],
    scriptSrc: ["'self'", "'unsafe-inline'", "'unsafe-eval'", 'https://cdn.redoc.ly'],
    fontSrc: ["'self'", 'https://fonts.gstatic.com'],
    imgSrc: ["'self'", 'data:', 'https:'],
    connectSrc: ["'self'"],
    workerSrc: ["'self'", 'blob:'],
  },
};
```

### 2. Nonce-Based Script Loading

Add nonce middleware:

```typescript
// src/middlewares/nonce.middleware.ts

import { Request, Response, NextFunction } from 'express';
import crypto from 'crypto';

export const nonceMiddleware = (req: Request, res: Response, next: NextFunction) => {
  // Generate unique nonce for this request
  const nonce = crypto.randomBytes(16).toString('base64');
  res.locals.nonce = nonce;

  // Will be used by Helmet CSP
  res.locals.cspNonce = nonce;

  next();
};

// Helper to get nonce in templates
export const getNonce = (res: Response): string => {
  return res.locals.nonce || '';
};
```

### 3. Update Helmet Configuration

```typescript
// In app.ts

import helmet from 'helmet';
import { strictCSP, docsCSP } from '@config/security';
import { nonceMiddleware } from '@middlewares/nonce.middleware';

// Add nonce to all requests
app.use(nonceMiddleware);

// Docs routes - relaxed CSP
app.use(['/docs', '/redoc', '/api-docs', '/developer'],
  helmet({
    contentSecurityPolicy: docsCSP,
  })
);

// All other routes - strict CSP with nonce
app.use(
  helmet({
    contentSecurityPolicy: {
      directives: {
        ...strictCSP.directives,
        scriptSrc: (req, res) => [
          "'self'",
          `'nonce-${res.locals.nonce}'`
        ],
        styleSrc: (req, res) => [
          "'self'",
          'https://fonts.googleapis.com',
          `'nonce-${res.locals.nonce}'`
        ],
      },
    },
    crossOriginEmbedderPolicy: true,
    crossOriginOpenerPolicy: { policy: 'same-origin' },
    crossOriginResourcePolicy: { policy: 'same-origin' },
    referrerPolicy: { policy: 'strict-origin-when-cross-origin' },
  })
);
```

### 4. Update View Templates

All inline scripts and styles need nonce attributes:

```typescript
// In view templates, pass nonce to scripts
<script nonce="${nonce}">
  // Inline script code
</script>

<style nonce="${nonce}">
  /* Inline styles */
</style>
```

### 5. Update View Generation Functions

Each view generator needs to accept and use nonce:

```typescript
// Example: src/views/landing/index.ts

interface LandingPageConfig {
  nonce: string;
  // ... other config
}

export const generateLandingPage = (config: LandingPageConfig): string => {
  const { nonce } = config;

  return `
    <!DOCTYPE html>
    <html>
    <head>
      <style nonce="${nonce}">
        ${getStyles()}
      </style>
    </head>
    <body>
      ${getContent()}
      <script nonce="${nonce}">
        ${getScripts()}
      </script>
    </body>
    </html>
  `;
};
```

### 6. Additional Security Headers

Add more security headers:

```typescript
// Additional headers beyond Helmet defaults
app.use((req, res, next) => {
  // Prevent MIME type sniffing
  res.setHeader('X-Content-Type-Options', 'nosniff');

  // Prevent clickjacking
  res.setHeader('X-Frame-Options', 'DENY');

  // XSS Protection (legacy, but still useful)
  res.setHeader('X-XSS-Protection', '1; mode=block');

  // Permissions Policy (formerly Feature Policy)
  res.setHeader('Permissions-Policy',
    'camera=(), microphone=(), geolocation=(), payment=()'
  );

  // Strict Transport Security (production only)
  if (process.env.NODE_ENV === 'production') {
    res.setHeader('Strict-Transport-Security',
      'max-age=31536000; includeSubDomains; preload'
    );
  }

  next();
});
```
