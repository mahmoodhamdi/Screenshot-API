### 1. Optimize Browser Pool

```typescript
// src/config/puppeteer.ts

interface BrowserPoolConfig {
  minBrowsers: number;
  maxBrowsers: number;
  maxPagesPerBrowser: number;
  browserIdleTimeout: number;
  pageIdleTimeout: number;
}

class BrowserPool {
  private browsers: Map<string, BrowserInstance> = new Map();
  private pageCount: Map<string, number> = new Map();
  private queue: Array<(browser: Browser) => void> = [];

  async getBrowser(): Promise<Browser> {
    // Find browser with capacity
    for (const [id, instance] of this.browsers) {
      const count = this.pageCount.get(id) || 0;
      if (count < this.config.maxPagesPerBrowser) {
        this.pageCount.set(id, count + 1);
        return instance.browser;
      }
    }

    // Create new browser if under limit
    if (this.browsers.size < this.config.maxBrowsers) {
      return this.createBrowser();
    }

    // Queue request
    return new Promise((resolve) => {
      this.queue.push(resolve);
    });
  }

  async releaseBrowser(browser: Browser): Promise<void> {
    const id = this.getBrowserId(browser);
    const count = this.pageCount.get(id) || 1;
    this.pageCount.set(id, count - 1);

    // Process queue
    if (this.queue.length > 0) {
      const resolve = this.queue.shift();
      resolve(browser);
    }
  }
}
```

### 2. Page Reuse for Similar Requests

```typescript
// Reuse pages for same viewport/settings
const pageCache = new Map<string, Page>();

const getPageCacheKey = (options: ScreenshotOptions): string => {
  return `${options.width}x${options.height}:${options.darkMode}:${options.blockAds}`;
};

const getOrCreatePage = async (browser: Browser, options: ScreenshotOptions): Promise<Page> => {
  const cacheKey = getPageCacheKey(options);
  const cached = pageCache.get(cacheKey);

  if (cached && !cached.isClosed()) {
    return cached;
  }

  const page = await browser.newPage();
  await setupPage(page, options);
  pageCache.set(cacheKey, page);

  return page;
};
```

### 3. Request Interception Optimization

```typescript
// Only intercept when needed (ads/images blocking)
const setupRequestInterception = async (page: Page, options: ScreenshotOptions) => {
  if (!options.blockAds && !options.blockImages) {
    return; // No interception needed
  }

  await page.setRequestInterception(true);

  const blockedPatterns = options.blockAds ? AD_PATTERNS : [];
  const blockedTypes = options.blockImages ? ['image'] : [];

  page.on('request', (request) => {
    const url = request.url();
    const type = request.resourceType();

    if (blockedTypes.includes(type)) {
      return request.abort();
    }

    if (blockedPatterns.some(p => p.test(url))) {
      return request.abort();
    }

    request.continue();
  });
};
```

### 4. Memory Management

```typescript
// Monitor and cleanup memory
const monitorBrowserMemory = async (browser: Browser) => {
  const pages = await browser.pages();
  let totalMemory = 0;

  for (const page of pages) {
    const metrics = await page.metrics();
    totalMemory += metrics.JSHeapUsedSize;
  }

  if (totalMemory > MAX_BROWSER_MEMORY) {
    logger.warn('Browser memory high, triggering cleanup', { totalMemory });
    await triggerGC(browser);
  }
};

// Force garbage collection
const triggerGC = async (browser: Browser) => {
  const pages = await browser.pages();
  for (const page of pages) {
    await page.evaluate(() => {
      if (window.gc) window.gc();
    });
  }
};
```
