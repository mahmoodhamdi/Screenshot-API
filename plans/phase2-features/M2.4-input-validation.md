## Context

The current input validation has some gaps:
- Weak password rules (only 8 chars, 1 uppercase, 1 lowercase, 1 digit)
- Custom headers in screenshot requests not sanitized
- Cookie values minimally validated
- URL schemes not fully validated

Create password validator:

```typescript
// src/utils/passwordValidator.ts

import zxcvbn from 'zxcvbn';

interface PasswordStrength {
  score: number; // 0-4
  feedback: string[];
  isStrong: boolean;
  crackTime: string;
}

// Common passwords to reject
const COMMON_PASSWORDS = new Set([
  'password', 'password123', '123456789', 'qwerty123',
  'admin123', 'letmein', 'welcome', 'monkey', 'dragon',
  '1234567890', 'password1', 'abc123', '111111', 'iloveyou'
]);

// Blocked patterns
const BLOCKED_PATTERNS = [
  /^(.)\1+$/,                    // All same character: aaaa
  /^(012|123|234|345|456|567|678|789)+$/, // Sequential numbers
  /^(abc|bcd|cde|def|efg|fgh|ghi|hij|ijk|jkl|klm|lmn|mno|nop|opq|pqr|qrs|rst|stu|tuv|uvw|vwx|wxy|xyz)+$/i, // Sequential letters
  /^(qwerty|asdf|zxcv)+$/i,     // Keyboard patterns
];

export const validatePasswordStrength = (
  password: string,
  userInputs: string[] = [] // email, name, etc. to check against
): PasswordStrength => {
  const feedback: string[] = [];

  // Check common passwords
  if (COMMON_PASSWORDS.has(password.toLowerCase())) {
    feedback.push('This is a commonly used password');
    return { score: 0, feedback, isStrong: false, crackTime: 'instant' };
  }

  // Check blocked patterns
  for (const pattern of BLOCKED_PATTERNS) {
    if (pattern.test(password)) {
      feedback.push('Password contains a predictable pattern');
      return { score: 0, feedback, isStrong: false, crackTime: 'instant' };
    }
  }

  // Run zxcvbn analysis
  const result = zxcvbn(password, userInputs);

  // Add zxcvbn feedback
  if (result.feedback.warning) {
    feedback.push(result.feedback.warning);
  }
  feedback.push(...result.feedback.suggestions);

  // Additional checks
  if (password.length < 12) {
    feedback.push('Consider using at least 12 characters');
  }

  if (!/[!@#$%^&*(),.?":{}|<>]/.test(password)) {
    feedback.push('Consider adding special characters');
  }

  // Score >= 3 is considered strong
  const isStrong = result.score >= 3;

  return {
    score: result.score,
    feedback,
    isStrong,
    crackTime: result.crack_times_display.offline_slow_hashing_1e4_per_second
  };
};

// Zod refine function
export const passwordStrengthRefine = (password: string): boolean => {
  const result = validatePasswordStrength(password);
  return result.isStrong;
};
```

### 2. Header Sanitization

```typescript
// src/utils/headerSanitizer.ts

// Headers that should never be overwritten
const FORBIDDEN_HEADERS = new Set([
  'host',
  'content-length',
  'transfer-encoding',
  'connection',
  'keep-alive',
  'upgrade',
  'proxy-authorization',
  'proxy-connection',
  'te',
  'trailer'
]);

// Headers that could be dangerous
const SENSITIVE_HEADERS = new Set([
  'authorization',
  'cookie',
  'set-cookie',
  'x-forwarded-for',
  'x-forwarded-host',
  'x-real-ip'
]);

interface SanitizeResult {
  headers: Record<string, string>;
  warnings: string[];
  blocked: string[];
}

export const sanitizeHeaders = (
  headers: Record<string, string>,
  options: {
    allowSensitive?: boolean;
    maxHeaders?: number;
    maxHeaderLength?: number;
  } = {}
): SanitizeResult => {
  const {
    allowSensitive = false,
    maxHeaders = 20,
    maxHeaderLength = 8192
  } = options;

  const result: SanitizeResult = {
    headers: {},
    warnings: [],
    blocked: []
  };

  const entries = Object.entries(headers);

  // Limit number of headers
  if (entries.length > maxHeaders) {
    result.warnings.push(`Too many headers (max ${maxHeaders}), truncating`);
  }

  for (const [key, value] of entries.slice(0, maxHeaders)) {
    const lowerKey = key.toLowerCase();

    // Block forbidden headers
    if (FORBIDDEN_HEADERS.has(lowerKey)) {
      result.blocked.push(key);
      continue;
    }

    // Warn about sensitive headers
    if (SENSITIVE_HEADERS.has(lowerKey) && !allowSensitive) {
      result.blocked.push(key);
      result.warnings.push(`Sensitive header blocked: ${key}`);
      continue;
    }

    // Validate header name (RFC 7230)
    if (!/^[a-zA-Z0-9\-_]+$/.test(key)) {
      result.blocked.push(key);
      result.warnings.push(`Invalid header name: ${key}`);
      continue;
    }

    // Validate header value (no CRLF injection)
    if (/[\r\n]/.test(value)) {
      result.blocked.push(key);
      result.warnings.push(`Header value contains invalid characters: ${key}`);
      continue;
    }

    // Truncate long values
    let sanitizedValue = value;
    if (value.length > maxHeaderLength) {
      sanitizedValue = value.substring(0, maxHeaderLength);
      result.warnings.push(`Header value truncated: ${key}`);
    }

    result.headers[key] = sanitizedValue;
  }

  return result;
};
```

### 3. Cookie Validation

```typescript
// src/utils/cookieSanitizer.ts

interface Cookie {
  name: string;
  value: string;
  domain?: string;
  path?: string;
  secure?: boolean;
  httpOnly?: boolean;
  sameSite?: 'Strict' | 'Lax' | 'None';
  expires?: number;
}

interface SanitizeResult {
  cookies: Cookie[];
  warnings: string[];
  blocked: string[];
}

// RFC 6265 compliant cookie name validation
const isValidCookieName = (name: string): boolean => {
  // Cookie names can't contain these characters
  const invalidChars = /[\x00-\x1f\x7f()<>@,;:\\"\/\[\]?={} \t]/;
  return name.length > 0 && name.length <= 256 && !invalidChars.test(name);
};

// Cookie value validation
const isValidCookieValue = (value: string): boolean => {
  // Cookie values can't contain these characters (unless quoted)
  const invalidChars = /[\x00-\x1f\x7f,;\\"]/;
  return value.length <= 4096 && !invalidChars.test(value);
};

// Domain validation
const isValidDomain = (domain: string, targetUrl: string): boolean => {
  try {
    const targetHost = new URL(targetUrl).hostname;
    const cookieDomain = domain.startsWith('.') ? domain.slice(1) : domain;

    // Cookie domain must be a suffix of the target domain
    return targetHost === cookieDomain || targetHost.endsWith('.' + cookieDomain);
  } catch {
    return false;
  }
};

export const sanitizeCookies = (
  cookies: Cookie[],
  targetUrl: string,
  options: {
    maxCookies?: number;
  } = {}
): SanitizeResult => {
  const { maxCookies = 50 } = options;

  const result: SanitizeResult = {
    cookies: [],
    warnings: [],
    blocked: []
  };

  if (cookies.length > maxCookies) {
    result.warnings.push(`Too many cookies (max ${maxCookies}), truncating`);
  }

  for (const cookie of cookies.slice(0, maxCookies)) {
    // Validate name
    if (!isValidCookieName(cookie.name)) {
      result.blocked.push(cookie.name);
      result.warnings.push(`Invalid cookie name: ${cookie.name}`);
      continue;
    }

    // Validate value
    if (!isValidCookieValue(cookie.value)) {
      result.blocked.push(cookie.name);
      result.warnings.push(`Invalid cookie value for: ${cookie.name}`);
      continue;
    }

    // Validate domain if specified
    if (cookie.domain && !isValidDomain(cookie.domain, targetUrl)) {
      result.blocked.push(cookie.name);
      result.warnings.push(`Cookie domain mismatch: ${cookie.name}`);
      continue;
    }

    result.cookies.push({
      name: cookie.name,
      value: cookie.value,
      domain: cookie.domain,
      path: cookie.path || '/',
      secure: cookie.secure,
      httpOnly: cookie.httpOnly,
      sameSite: cookie.sameSite,
      expires: cookie.expires
    });
  }

  return result;
};
```

### 4. Enhanced URL Validation

```typescript
// src/utils/urlValidator.ts

interface UrlValidationResult {
  valid: boolean;
  url?: string;
  error?: string;
  warnings: string[];
}

// Blocked URL schemes
const BLOCKED_SCHEMES = new Set([
  'javascript',
  'data',
  'file',
  'vbscript',
  'about',
  'blob',
  'chrome',
  'chrome-extension'
]);

// Blocked file extensions (potential security risk)
const BLOCKED_EXTENSIONS = new Set([
  '.exe', '.bat', '.cmd', '.sh', '.ps1',
  '.dll', '.so', '.dylib'
]);

export const validateScreenshotUrl = (url: string): UrlValidationResult => {
  const warnings: string[] = [];

  try {
    const parsed = new URL(url);

    // Check scheme
    const scheme = parsed.protocol.replace(':', '').toLowerCase();

    if (BLOCKED_SCHEMES.has(scheme)) {
      return {
        valid: false,
        error: `URL scheme '${scheme}' is not allowed`,
        warnings
      };
    }

    // Only allow http and https
    if (!['http', 'https'].includes(scheme)) {
      return {
        valid: false,
        error: `URL scheme '${scheme}' is not supported. Use http or https.`,
        warnings
      };
    }

    // Warn about HTTP (prefer HTTPS)
    if (scheme === 'http') {
      warnings.push('HTTP URLs may not work correctly on sites with HTTPS redirects');
    }

    // Check for blocked file extensions
    const pathname = parsed.pathname.toLowerCase();
    for (const ext of BLOCKED_EXTENSIONS) {
      if (pathname.endsWith(ext)) {
        return {
          valid: false,
          error: `URL points to a blocked file type: ${ext}`,
          warnings
        };
      }
    }

    // Check hostname
    const hostname = parsed.hostname.toLowerCase();

    // Block localhost and internal addresses
    const internalPatterns = [
      'localhost',
      '127.0.0.1',
      '0.0.0.0',
      '::1',
      '[::1]',
      /^10\.\d+\.\d+\.\d+$/,
      /^172\.(1[6-9]|2\d|3[0-1])\.\d+\.\d+$/,
      /^192\.168\.\d+\.\d+$/,
      /^169\.254\.\d+\.\d+$/,
      /^fc00:/,
      /^fe80:/
    ];

    for (const pattern of internalPatterns) {
      if (typeof pattern === 'string' ? hostname === pattern : pattern.test(hostname)) {
        return {
          valid: false,
          error: 'Internal/private URLs are not allowed',
          warnings
        };
      }
    }

    // Check for suspicious ports
    if (parsed.port) {
      const port = parseInt(parsed.port);
      const suspiciousPorts = [22, 23, 25, 110, 143, 3389, 5900];

      if (suspiciousPorts.includes(port)) {
        warnings.push(`Unusual port ${port} may indicate non-web service`);
      }
    }

    // Check for excessively long URL
    if (url.length > 2048) {
      warnings.push('Very long URL may cause issues');
    }

    return {
      valid: true,
      url: parsed.toString(),
      warnings
    };
  } catch (error) {
    return {
      valid: false,
      error: 'Invalid URL format',
      warnings
    };
  }
};
```

### 6. Add Password Strength Endpoint

```typescript
// In auth.routes.ts

// Check password strength (for UI feedback)
router.post('/check-password-strength',
  defaultRateLimit,
  async (req, res) => {
    const { password, email, name } = req.body;

    if (!password) {
      return res.status(400).json({
        success: false,
        error: 'Password required'
      });
    }

    const userInputs = [email, name].filter(Boolean);
    const result = validatePasswordStrength(password, userInputs);

    res.json({
      success: true,
      data: {
        score: result.score,
        feedback: result.feedback,
        isStrong: result.isStrong,
        crackTime: result.crackTime
      }
    });
  }
);
```
