# Milestone 2.3: Webhook Security

## Status: NOT_STARTED
## Priority: HIGH
## Estimated Effort: Medium

---

## Context

The webhook delivery system for screenshots has security issues:
- No HTTPS enforcement for webhook URLs
- No signature verification for outgoing webhooks
- Fire-and-forget delivery with limited retry
- No persistence of webhook attempts

---

## Current State

### Current Implementation (screenshot.service.ts)

```typescript
// Lines 575-627 - sendWebhook function
private async sendWebhook(url: string, payload: object, retries = 3): Promise<void> {
  for (let attempt = 1; attempt <= retries; attempt++) {
    try {
      await fetch(url, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(payload)
      });
      return;
    } catch (error) {
      if (attempt === retries) {
        logger.error('Webhook delivery failed', { url, error });
      }
      await new Promise(r => setTimeout(r, attempt * 1000));
    }
  }
}
```

### Problems
1. Accepts `http://` URLs (data exposure)
2. No signature for verification by receiver
3. Only 3 retries, no persistence
4. No jitter in backoff (thundering herd)
5. No dead letter queue

```typescript
// In src/middlewares/validation.middleware.ts

const validateWebhookUrl = (url: string): boolean => {
  if (!url) return true; // Optional field

  try {
    const parsed = new URL(url);

    // Require HTTPS in production
    if (config.nodeEnv === 'production' && parsed.protocol !== 'https:') {
      return false;
    }

    // Block internal/private IPs
    const hostname = parsed.hostname.toLowerCase();
    const blockedPatterns = [
      'localhost',
      '127.0.0.1',
      '0.0.0.0',
      '::1',
      /^10\./,
      /^172\.(1[6-9]|2[0-9]|3[0-1])\./,
      /^192\.168\./,
      /^169\.254\./
    ];

    if (blockedPatterns.some(p =>
      typeof p === 'string' ? hostname === p : p.test(hostname)
    )) {
      return false;
    }

    return true;
  } catch {
    return false;
  }
};

// Update createScreenshotSchema
export const createScreenshotSchema = z.object({

    .refine(validateWebhookUrl, {

        : 'Webhook URL cannot target internal addresses'
    })
});
```

### 2. Webhook Signature

```typescript
// src/utils/webhookSignature.ts

import crypto from 'crypto';
import { config } from '@config/index';

interface SignedPayload {
  payload: object;
  timestamp: number;
  signature: string;
}

export const createWebhookSignature = (
  payload: object,
  secret: string,
  timestamp: number = Date.now()
): SignedPayload => {
  const payloadString = JSON.stringify(payload);
  const signaturePayload = `${timestamp}.${payloadString}`;

  const signature = crypto
    .createHmac('sha256', secret)
    .update(signaturePayload)
    .digest('hex');

  return {
    payload,
    timestamp,
    signature: `sha256=${signature}`
  };
};

export const verifyWebhookSignature = (
  payload: string,
  timestamp: number,
  signature: string,
  secret: string,
  toleranceMs: number = 300000 // 5 minutes
): boolean => {
  // Check timestamp freshness
  const now = Date.now();
  if (Math.abs(now - timestamp) > toleranceMs) {
    return false;
  }

  const signaturePayload = `${timestamp}.${payload}`;
  const expectedSignature = crypto
    .createHmac('sha256', secret)
    .update(signaturePayload)
    .digest('hex');

  return crypto.timingSafeEqual(
    Buffer.from(signature.replace('sha256=', '')),
    Buffer.from(expectedSignature)
  );
};
```

### 4. Webhook Queue with Bull

```typescript
// src/queues/webhook.queue.ts

import Bull from 'bull';
import { config } from '@config/index';
import { WebhookAttempt } from '@models/webhookAttempt.model';
import { createWebhookSignature } from '@utils/webhookSignature';
import { logger } from '@utils/logger';

interface WebhookJob {
  attemptId: string;
}

const webhookQueue = new Bull<WebhookJob>('webhooks', {
  redis: {
    host: config.redis.host,
    port: config.redis.port,
    password: config.redis.password
  },
  defaultJobOptions: {
    attempts: 1, // We handle retries via the attempt model
    removeOnComplete: 100,
    removeOnFail: 500
  }
});

// Calculate delay with exponential backoff + jitter
const calculateDelay = (attempt: number): number => {
  const baseDelay = 1000; // 1 second
  const maxDelay = 3600000; // 1 hour
  const exponentialDelay = baseDelay * Math.pow(2, attempt - 1);
  const jitter = Math.random() * 0.3 * exponentialDelay; // 0-30% jitter
  return Math.min(exponentialDelay + jitter, maxDelay);
};

webhookQueue.process(async (job) => {
  const { attemptId } = job.data;

  const attempt = await WebhookAttempt.findById(attemptId);
  if (!attempt || attempt.status !== 'pending') {
    return { skipped: true };
  }

  attempt.attempts += 1;
  attempt.lastAttemptAt = new Date();

  try {
    // Get user's webhook secret
    const user = await User.findById(attempt.userId);
    const webhookSecret = user?.webhookSecret || config.defaultWebhookSecret;

    // Create signed payload
    const signed = createWebhookSignature(attempt.payload, webhookSecret);

    const response = await fetch(attempt.url, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'X-Webhook-Timestamp': signed.timestamp.toString(),
        'X-Webhook-Signature': signed.signature,
        'X-Webhook-ID': attempt._id.toString(),
        'User-Agent': 'ScreenshotAPI-Webhook/1.0'
      },
      body: JSON.stringify(signed.payload),
      signal: AbortSignal.timeout(30000) // 30 second timeout
    });

    attempt.responseStatus = response.status;

    // Read response body (limited)
    const body = await response.text();
    attempt.responseBody = body.substring(0, 1000);

    if (response.ok) {
      attempt.status = 'success';
      logger.info('Webhook delivered', {
        attemptId,
        url: attempt.url,
        statusCode: response.status
      });
    } else {
      throw new Error(`HTTP ${response.status}`);
    }
  } catch (error) {
    const errorMessage = error instanceof Error ? error.message : 'Unknown error';
    attempt.errorMessage = errorMessage;

    if (attempt.attempts >= attempt.maxAttempts) {
      attempt.status = 'failed';
      logger.error('Webhook permanently failed', {
        attemptId,
        url: attempt.url,
        attempts: attempt.attempts,
        error: errorMessage
      });
    } else {
      // Schedule retry
      const delay = calculateDelay(attempt.attempts);
      attempt.nextAttemptAt = new Date(Date.now() + delay);

      // Add retry job
      await webhookQueue.add(
        { attemptId },
        { delay }
      );

      logger.warn('Webhook failed, scheduling retry', {
        attemptId,
        attempt: attempt.attempts,
        nextAttemptIn: Math.round(delay / 1000) + 's'
      });
    }
  }

  await attempt.save();
  return { success: attempt.status === 'success' };
});

// Queue a new webhook
export const queueWebhook = async (data: {
  screenshotId: string;
  userId: string;
  url: string;
  payload: object;
}): Promise<string> => {
  const attempt = await WebhookAttempt.create({
    screenshotId: data.screenshotId,
    userId: data.userId,
    url: data.url,
    payload: data.payload,
    nextAttemptAt: new Date()
  });

  await webhookQueue.add({ attemptId: attempt._id.toString() });

  return attempt._id.toString();
};

// Get webhook status
export const getWebhookStatus = async (attemptId: string) => {
  return WebhookAttempt.findById(attemptId);
};

// Retry a failed webhook manually
export const retryWebhook = async (attemptId: string): Promise<boolean> => {
  const attempt = await WebhookAttempt.findById(attemptId);
  if (!attempt || attempt.status !== 'failed') {
    return false;
  }

  attempt.status = 'pending';
  attempt.attempts = 0;
  attempt.nextAttemptAt = new Date();
  await attempt.save();

  await webhookQueue.add({ attemptId });
  return true;
};

export { webhookQueue };
```

### 7. Add API Endpoints

```typescript
// In routes/webhooks.routes.ts

import { Router } from 'express';
import { authenticateAny } from '@middlewares/auth.middleware';

const router = Router();

// Get webhook history
router.get('/webhooks',
  authenticateAny,
  async (req, res) => {
    const attempts = await WebhookAttempt.find({ userId: req.user._id })
      .sort({ createdAt: -1 })
      .limit(100);

    res.json({ success: true, data: attempts });
  }
);

// Get webhook details
router.get('/webhooks/:id',
  authenticateAny,
  async (req, res) => {
    const attempt = await WebhookAttempt.findOne({
      _id: req.params.id,
      userId: req.user._id
    });

    if (!attempt) {
      return res.status(404).json({ success: false, error: 'Webhook not found' });
    }

    res.json({ success: true, data: attempt });
  }
);

// Retry failed webhook
router.post('/webhooks/:id/retry',
  authenticateAny,
  async (req, res) => {
    const success = await retryWebhook(req.params.id);

    if (!success) {
      return res.status(400).json({
        success: false,
        error: 'Cannot retry webhook (not found or not failed)'
      });
    }

    res.json({ success: true, message: 'Webhook retry queued' });
  }
);

// Get webhook secret
router.get('/webhook-secret',
  authenticateAny,
  async (req, res) => {
    const user = await User.findById(req.user._id).select('+webhookSecret');
    res.json({ success: true, data: { secret: user?.webhookSecret } });
  }
);

// Regenerate webhook secret
router.post('/webhook-secret/regenerate',
  authenticateAny,
  async (req, res) => {
    const user = await User.findById(req.user._id).select('+webhookSecret');
    const secret = await user?.regenerateWebhookSecret();
    res.json({ success: true, data: { secret } });
  }
);

export default router;
```
