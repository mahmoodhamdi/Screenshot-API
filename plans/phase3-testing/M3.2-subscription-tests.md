```typescript
describe('Subscription API Endpoints', () => {
  describe('GET /api/v1/subscriptions/plans', () => {
    it('should return all available plans');
    it('should include pricing for each plan');
    it('should include features for each plan');
    it('should not require authentication');
  });

  describe('POST /api/v1/subscriptions/checkout', () => {
    it('should require authentication');
    it('should create Stripe checkout session');
    it('should return checkout URL');
    it('should validate plan parameter');
    it('should validate success/cancel URLs');
    it('should reject if already on plan');
  });

  describe('POST /api/v1/subscriptions/portal', () => {
    it('should require authentication');
    it('should create customer portal session');
    it('should return portal URL');
    it('should create Stripe customer if not exists');
  });

  describe('POST /api/v1/subscriptions/webhook', () => {
    it('should verify Stripe signature');
    it('should reject invalid signature');
    it('should handle checkout.session.completed');
    it('should handle customer.subscription.updated');
    it('should handle customer.subscription.deleted');
    it('should handle invoice.payment_succeeded');
    it('should handle invoice.payment_failed');
    it('should update user subscription status');
    it('should reset usage on new billing period');
  });

  describe('GET /api/v1/subscriptions', () => {
    it('should return current subscription');
    it('should include plan details');
    it('should include billing period');
    it('should include next billing date');
  });

  describe('PUT /api/v1/subscriptions/plan', () => {
    it('should upgrade plan');
    it('should downgrade plan');
    it('should handle proration');
    it('should update user plan immediately');
  });

  describe('DELETE /api/v1/subscriptions', () => {
    it('should cancel subscription');
    it('should set cancel_at_period_end');
    it('should downgrade to free at period end');
  });

  describe('POST /api/v1/subscriptions/resume', () => {
    it('should resume cancelled subscription');
    it('should clear cancel_at_period_end');
  });

  describe('GET /api/v1/subscriptions/usage', () => {
    it('should return usage statistics');
    it('should include screenshots used/limit');
    it('should include API calls');
    it('should include days until reset');
  });
});
```
