## Test Infrastructure

### Mock Webhook Server

```typescript
// tests/helpers/mockWebhookServer.ts
import express from 'express';
import { verifyWebhookSignature } from '../../src/utils/webhookSignature';

export const startMockWebhookServer = async (port: number) => {
  const app = express();
  const receivedWebhooks: any[] = [];

  app.use(express.raw({ type: 'application/json' }));

  app.post('/webhook', (req, res) => {
    const payload = req.body.toString();
    const timestamp = req.headers['x-webhook-timestamp'];
    const signature = req.headers['x-webhook-signature'];

    receivedWebhooks.push({
      payload: JSON.parse(payload),
      timestamp,
      signature,
      valid: verifyWebhookSignature(payload, timestamp, signature, SECRET)
    });

    res.status(200).send('OK');
  });

  const server = app.listen(port);

  return {
    server,
    receivedWebhooks,
    close: () => new Promise(resolve => server.close(resolve))
  };
};
```
