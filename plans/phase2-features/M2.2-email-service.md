```typescript
// src/services/email.service.ts

import nodemailer, { Transporter } from 'nodemailer';
import { config } from '@config/index';
import { logger } from '@utils/logger';

interface EmailOptions {
  to: string;
  subject: string;
  html: string;
  text?: string;
}

interface EmailResult {
  success: boolean;
  messageId?: string;
  error?: string;
}

class EmailService {
  private transporter: Transporter | null = null;
  private initialized: boolean = false;

  async initialize(): Promise<void> {
    if (this.initialized) return;

    try {
      this.transporter = nodemailer.createTransport({
        host: config.email.host,
        port: config.email.port,
        secure: config.email.secure,
        auth: {
          user: config.email.user,
          pass: config.email.password
        }
      });

      // Verify connection
      await this.transporter.verify();
      this.initialized = true;
      logger.info('Email service initialized');
    } catch (error) {
      logger.error('Failed to initialize email service', { error });
      throw error;
    }
  }

  async send(options: EmailOptions): Promise<EmailResult> {
    if (!this.transporter) {
      await this.initialize();
    }

    try {
      const result = await this.transporter!.sendMail({
        from: `"${config.email.fromName}" <${config.email.fromAddress}>`,
        to: options.to,
        subject: options.subject,
        html: options.html,
        text: options.text || this.htmlToText(options.html)
      });

      logger.info('Email sent', {
        to: options.to,
        subject: options.subject,
        messageId: result.messageId
      });

      return { success: true, messageId: result.messageId };
    } catch (error) {
      logger.error('Failed to send email', {
        to: options.to,
        subject: options.subject,
        error
      });

      return {
        success: false,
        error: error instanceof Error ? error.message : 'Unknown error'
      };
    }
  }

  private htmlToText(html: string): string {
    return html
      .replace(/<[^>]*>/g, '')
      .replace(/\s+/g, ' ')
      .trim();
  }

  // Specific email methods
  async sendPasswordResetEmail(email: string, token: string): Promise<EmailResult> {
    const resetUrl = `${config.baseUrl}/reset-password?token=${token}`;

    return this.send({
      to: email,
      subject: 'Reset Your Password - Screenshot API',
      html: emailTemplates.passwordReset({ resetUrl, expiresIn: '1 hour' })
    });
  }

  async sendVerificationEmail(email: string, token: string): Promise<EmailResult> {
    const verifyUrl = `${config.baseUrl}/verify-email?token=${token}`;

    return this.send({
      to: email,
      subject: 'Verify Your Email - Screenshot API',
      html: emailTemplates.emailVerification({ verifyUrl })
    });
  }

  async sendWelcomeEmail(email: string, name: string): Promise<EmailResult> {
    return this.send({
      to: email,
      subject: 'Welcome to Screenshot API!',
      html: emailTemplates.welcome({ name })
    });
  }

  async sendPaymentFailedEmail(email: string, details: {
    planName: string;
    amount: string;
    retryDate?: string;
  }): Promise<EmailResult> {
    return this.send({
      to: email,
      subject: 'Payment Failed - Screenshot API',
      html: emailTemplates.paymentFailed(details)
    });
  }

  async sendSubscriptionChangedEmail(email: string, details: {
    oldPlan: string;
    newPlan: string;
    effectiveDate: string;
  }): Promise<EmailResult> {
    return this.send({
      to: email,
      subject: 'Subscription Updated - Screenshot API',
      html: emailTemplates.subscriptionChanged(details)
    });
  }

  async sendInvoiceEmail(email: string, details: {
    invoiceNumber: string;
    amount: string;
    date: string;
    downloadUrl: string;
  }): Promise<EmailResult> {
    return this.send({
      to: email,
      subject: `Invoice ${details.invoiceNumber} - Screenshot API`,
      html: emailTemplates.invoice(details)
    });
  }
}

export const emailService = new EmailService();
```

### 3. Create Email Queue with Bull

```typescript
// src/queues/email.queue.ts

import Bull from 'bull';
import { config } from '@config/index';
import { emailService } from '@services/email.service';
import { logger } from '@utils/logger';

interface EmailJob {
  type: 'passwordReset' | 'verification' | 'welcome' | 'paymentFailed' | 'subscriptionChanged' | 'invoice';
  to: string;
  data: Record<string, unknown>;
}

const emailQueue = new Bull<EmailJob>('email', {
  redis: {
    host: config.redis.host,
    port: config.redis.port,
    password: config.redis.password
  },
  defaultJobOptions: {
    attempts: 3,
    backoff: {
      type: 'exponential',
      delay: 2000
    },
    removeOnComplete: 100,
    removeOnFail: 500
  }
});

// Process jobs
emailQueue.process(async (job) => {
  const { type, to, data } = job.data;

  logger.info('Processing email job', { type, to, jobId: job.id });

  let result;

  switch (type) {
    case 'passwordReset':
      result = await emailService.sendPasswordResetEmail(to, data.token as string);
      break;
    case 'verification':
      result = await emailService.sendVerificationEmail(to, data.token as string);
      break;
    case 'welcome':
      result = await emailService.sendWelcomeEmail(to, data.name as string);
      break;
    case 'paymentFailed':
      result = await emailService.sendPaymentFailedEmail(to, data as any);
      break;
    case 'subscriptionChanged':
      result = await emailService.sendSubscriptionChangedEmail(to, data as any);
      break;
    case 'invoice':
      result = await emailService.sendInvoiceEmail(to, data as any);
      break;
    default:
      throw new Error(`Unknown email type: ${type}`);
  }

  if (!result.success) {
    throw new Error(result.error || 'Email send failed');
  }

  return result;
});

// Event handlers
emailQueue.on('completed', (job, result) => {
  logger.info('Email job completed', {
    jobId: job.id,
    type: job.data.type,
    messageId: result.messageId
  });
});

emailQueue.on('failed', (job, error) => {
  logger.error('Email job failed', {
    jobId: job.id,
    type: job.data.type,
    error: error.message,
    attempts: job.attemptsMade
  });
});

// Queue helper functions
export const queueEmail = {
  passwordReset: (email: string, token: string) =>
    emailQueue.add({ type: 'passwordReset', to: email, data: { token } }),

  verification: (email: string, token: string) =>
    emailQueue.add({ type: 'verification', to: email, data: { token } }),

  welcome: (email: string, name: string) =>
    emailQueue.add({ type: 'welcome', to: email, data: { name } }),

  paymentFailed: (email: string, data: { planName: string; amount: string; retryDate?: string }) =>
    emailQueue.add({ type: 'paymentFailed', to: email, data }),

  subscriptionChanged: (email: string, data: { oldPlan: string; newPlan: string; effectiveDate: string }) =>
    emailQueue.add({ type: 'subscriptionChanged', to: email, data }),

  invoice: (email: string, data: { invoiceNumber: string; amount: string; date: string; downloadUrl: string }) =>
    emailQueue.add({ type: 'invoice', to: email, data })
};

export { emailQueue };
```
