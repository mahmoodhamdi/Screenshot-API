  const plainKey = `ss_test_${crypto.randomBytes(24).toString('hex')}`;

  const apiKey = await ApiKey.create({
    user: userId,
    name: 'Test Key',
    keyHash: hashedKey,
    keyPrefix: plainKey.substring(0, 10),
    permissions: ['read', 'write', 'delete']
  });

  return { apiKey, plainKey };
};
