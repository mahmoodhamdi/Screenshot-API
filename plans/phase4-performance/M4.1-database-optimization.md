### 3. Cursor-Based Pagination

```typescript
// Replace skip/limit with cursor pagination
interface CursorPaginationResult<T> {
  data: T[];
  nextCursor: string | null;
  prevCursor: string | null;
  hasMore: boolean;
}

const paginateWithCursor = async <T>(
  model: Model<T>,
  query: FilterQuery<T>,
  cursor: string | null,
  limit: number,
  sortField: string = 'createdAt'
): Promise<CursorPaginationResult<T>> => {
  const filter = cursor
    ? { ...query, [sortField]: { $lt: decodeCursor(cursor) } }
    : query;

  const docs = await model
    .find(filter)
    .sort({ [sortField]: -1 })
    .limit(limit + 1)
    .lean();

  const hasMore = docs.length > limit;
  const data = hasMore ? docs.slice(0, -1) : docs;

  return {
    data,
    nextCursor: hasMore ? encodeCursor(data[data.length - 1][sortField]) : null,
    prevCursor: cursor,
    hasMore
  };
};
```
