// Allow clients to select specific fields
router.get('/screenshots',
  validateQuery(z.object({
    fields: z.string().optional() // comma-separated fields
  })),
  async (req, res) => {
    const fields = req.query.fields?.split(',').join(' ') || '';
    const screenshots = await Screenshot.find({ user: req.user._id })

### 2. Compression Tuning

```typescript
// In app.ts
import compression from 'compression';

app.use(compression({
  level: 6, // Balance between speed and compression
  threshold: 1024, // Only compress responses > 1KB
  filter: (req, res) => {
    // Don't compress images/PDFs (already compressed)
    const contentType = res.getHeader('Content-Type');
    if (contentType?.includes('image/') || contentType?.includes('pdf')) {
      return false;
    }
    return compression.filter(req, res);
  }
}));
```

### 3. ETags for Caching

```typescript
import etag from 'etag';

// Add ETag middleware
const etagMiddleware = (req, res, next) => {
  const originalJson = res.json.bind(res);

  res.json = (data) => {
    const body = JSON.stringify(data);
    const etagValue = etag(body);

    res.setHeader('ETag', etagValue);

    if (req.headers['if-none-match'] === etagValue) {
      return res.status(304).end();
    }

    res.setHeader('Cache-Control', 'private, max-age=0, must-revalidate');
    return originalJson(data);
  };

  next();
};
```

### 4. Response Time Tracking

```typescript
// Track response times
const responseTimeMiddleware = (req, res, next) => {
  const start = process.hrtime.bigint();

  res.on('finish', () => {
    const end = process.hrtime.bigint();
    const durationMs = Number(end - start) / 1e6;

    res.setHeader('X-Response-Time', `${durationMs.toFixed(2)}ms`);

    // Log slow requests
    if (durationMs > 1000) {
      logger.warn('Slow request', {
        path: req.path,
        method: req.method,
        duration: durationMs
      });
    }
  });

  next();
};
```
